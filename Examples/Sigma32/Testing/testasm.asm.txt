; RRR instructions

   add    R8,R12,R5   ; 08c5

   addc   R8,R12,R5   ; 58c5
   muln   R8,R12,R5   ; 68c5
   divn   R8,R12,R5   ; 78c5
   trap   R8,R12,R5   ; c8c5
